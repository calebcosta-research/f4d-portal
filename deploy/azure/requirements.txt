# --- Database drivers -------------------------------------------------------
# psycopg2-binary: Azure Database for PostgreSQL (db_backend=postgres). A
# self-contained wheel, so it needs nothing from the system image.
psycopg2-binary==2.9.9
# pymssql: Azure SQL / SQL Server (db_backend=mssql, sql_driver=pymssql). Also
# self-contained. NOTE it cannot do Entra ID / managed-identity auth -- it is
# username+password only. If the architecture mandates managed identity to the
# database, pyodbc is required instead, and then the msodbcsql18 system driver
# has to be installed via a custom container or a startup script, because the
# stock App Service Python image does not ship it.
pymssql==2.3.1
